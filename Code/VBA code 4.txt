Sub CreatePieChart(wsSummary As Worksheet, wsGraphs As Worksheet, tableStartRow As Long, chartTitle As String)

    Dim lastRow As Long
    lastRow = wsSummary.Cells(wsSummary.Rows.Count, 1).End(xlUp).Row

    Dim cht As Chart
    Set cht = wsGraphs.Shapes.AddChart2(227, xlPie).Chart

    With cht
        .SetSourceData wsSummary.Range("A" & tableStartRow + 1 & ":B" & lastRow)
        .HasTitle = True
        .ChartTitle.Text = chartTitle
    End With

End Sub
